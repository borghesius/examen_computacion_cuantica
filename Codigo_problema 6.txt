# 1. Cargar estado cuantico
# -------------------------
def cargar_estado(nombre_archivo, N):
    psi = np.zeros(2**N, dtype=complex)

    with open(nombre_archivo, 'r') as f:
        for linea in f:
            if ':' not in linea:
                continue

            izquierda, derecha = linea.split(':')
            indice_bin = izquierda.strip()

            valores = derecha.strip().split()
            parte_real = float(valores[0])

            indice = int(indice_bin, 2)
            psi[indice] = parte_real

    # normalizar
    psi = psi / np.linalg.norm(psi)

    return psi

# ------------------------------
# 2. Entropía de entrelazamiento
# ------------------------------
def entropia_entrelazamiento(psi, N, bloque):
    """
    bloque: lista de qubits en el bloque A
    """
    dimensiones = [2]*N
    psi_tensor = psi.reshape(dimensiones)

    # Permutar indices
    A = list(bloque)
    B = [i for i in range(N) if i not in A]
    perm = A + B

    psi_perm = np.transpose(psi_tensor, perm)

    dimA = 2**len(A)
    dimB = 2**len(B)

    psi_matriz = psi_perm.reshape(dimA, dimB)

    # matriz reducida
    rhoA = psi_matriz @ psi_matriz.conj().T

    # autovalores
    autovalores = np.linalg.eigvalsh(rhoA)
    autovalores = autovalores[autovalores > 1e-12]

    # entropía de Von Neumann
    S = -np.sum(autovalores * np.log(autovalores))
    return S

# ---------------------
# 3. Construir matriz A
# ---------------------
def construir_matriz_A(N):
    subconjuntos = []
    for i in range(2**N):
        bits = [(i >> k) & 1 for k in range(N)]
        subconjuntos.append(bits)

    pares = list(combinations(range(N), 2))
    A = np.zeros((2**N, len(pares)))

    for l, x in enumerate(subconjuntos):
        for p, (i, j) in enumerate(pares):
            if (x[i], x[j]) in [(0,1), (1,0)]:
                A[l, p] = 1

    return A, pares

# --------------------
# 4. Calcular vector S
# --------------------
def calcular_S(psi, N):
    S = np.zeros(2**N)

    for l in range(2**N):
        bloque = [i for i in range(N) if (l >> i) & 1]
        if len(bloque) == 0 or len(bloque) == N:
            S[l] = 0
        else:
            S[l] = entropia_entrelazamiento(psi, N, bloque)

    return S

# ----------------------
# 5. Resolver ecuacion J
# ----------------------
def resolver_J(A, S):
    ATA = A.T @ A
    ATS = A.T @ S

    # Resolver sistema lineal
    #J = np.linalg.solve(ATA, ATS)
    J, _ = nnls(ATA, ATS)
    return J

# ----------------------------------
# 6. Convertir vector J a matriz NxN
# ----------------------------------
def J_a_matriz(J, pares, N):
    M = np.zeros((N, N))

    for valor, (i, j) in zip(J, pares):
        M[i, j] = valor
        M[j, i] = valor

    return M

# ----------------------
# 7. Geometria emergente
# ----------------------
def geometria_emergente(J, pares, N):
    J_max = math.log(2)
    J_geometria = [(i/J_max)**-0.5 if i > 0 else -1 for i in J]
    D = np.zeros((N, N))

    for valor, (i, j) in zip(J_geometria, pares):
        D[i, j] = valor
        D[j, i] = valor

    return D
